From apps/web/lib/actions/planta.ts (presigned URL pattern):
```typescript
export async function getPresignedUploadUrl(filename: string, contentType: string) {
  const { empresa } = await getTenantContext()
  const r2 = getR2Client()
  const key = `plantas/${empresa.tenantId}/${Date.now()}-${filename}`
  const url = await getSignedUrl(r2, new PutObjectCommand({
    Bucket: process.env.R2_BUCKET_NAME!,
    Key: key,
    ContentType: contentType,
  }), { expiresIn: 300 })
  return { url, key }
}
```

```typescript
tc.tarefa.findFirst({
  where: { obraId, status: "EM_ANDAMENTO", startDate: { lte: date }, endDate: { gte: date } },
  orderBy: { startDate: "asc" },
  include: { etapa: true },
})
```

From 03-RESEARCH.md — Progress calculation:
```
progressoReal = COUNT(DISTINCT DATE(RDO.data)) / durationDays * 100
```
